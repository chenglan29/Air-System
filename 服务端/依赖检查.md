# 依赖检查
依赖信息储存在每个应用的索引文件(index.json)的dependentSoftwares项中
例如:
```json
{
    "name":"FM",
    "developer":"橙蓝",
    "describe":"",
    "versionNumber":4,
    "versionName":"0.4",
    "dependentSoftwares":[
        {
        "name":"Info",
        "developer":"橙蓝",
        "describe":"",
        "versionNumber":7,
        "versionName":"0.7",
        "dependentSoftwares":[],
        "AvailableSystemVersions":[13],
        "UnavailableSystemVersions":[1,2,3,4,5,6],
        "priority":0,
        "event":{
            "closeConnection":[],
            "createConnection":[],
            "getMessage":[],
            "getMessage_doLogin":[],
            "initServer":["./software/Info/initServer.js"],
            "login":["./software/Info/login.js"],
            "ptp":[]
        }
    }],
    "AvailableSystemVersions":[13],
    "UnavailableSystemVersions":[1,2,3,4,5,6,7,8],
    "priority":0,
    "event":{
        "closeConnection":[],
        "createConnection":[],
        "getMessage":[],
        "getMessage_doLogin":[],
        "initServer":["./software/FM/init.js"],
        "login":["./software/FM/login.js"],
        "ptp":["./software/FM/ptp.js"]
    }
}
```
这里声明 FM-橙蓝(0.4) 依赖 Info-橙蓝(0.7) 一般的，我们引用依赖信息时只保留名字，开发者，版本号等信息
如:
```json
{
    "name":"FM",
    "developer":"橙蓝",
    "describe":"",
    "versionNumber":4,
    "versionName":"0.4",
    "dependentSoftwares":[
        {
            "name":"Info",
            "developer":"橙蓝",
            "versionNumber":7
        }
    ],
    "AvailableSystemVersions":[13],
    "UnavailableSystemVersions":[1,2,3,4,5,6,7,8],
    "priority":0,
    "event":{
        "closeConnection":[],
        "createConnection":[],
        "getMessage":[],
        "getMessage_doLogin":[],
        "initServer":["./software/FM/init.js"],
        "login":["./software/FM/login.js"],
        "ptp":["./software/FM/ptp.js"]
    }
}
```
若将版本号一并省略，则表示任一版本的info均可
